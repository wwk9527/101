```
def max_money(n, m, prices):
    prices.sort() # 对价格列表进行升序排序
    result = 0
    for price in prices[:m]:
        if price<=0:#虑前m个价格
             result -= price
    return result
 
n, m = map(int, input().split())
prices = list(map(int, input().split()))
print(max_money(n, m, prices))
```

### 158B. Taxi



*special problem, greedy, implementation, 1100, https://codeforces.com/problemset/problem/158/B

思路：

代码

```
def main():
    n = int(input())
    a = list(map(int, input().split()))
 
    # 按降序排序
    a.sort(reverse=True)
 
    ans = 0
    while a:
        tmp = a[0]  # 取出当前最大的元素
        a.pop(0)  # 移除当前最大的元素
        n -= 1  # 减少乘客数量
 
        # 尽可能多地添加乘客，直到总和超过4
        while a and tmp + a[-1] <= 4:
            tmp += a[-1]
            a.pop()  # 移除最后一个元素
            n -= 1  # 减少乘客数量
 
        ans += 1  # 增加一辆出租车的数量
 
    print(ans)
 
 
# 调用函数
if __name__ == "__main__":
    main()

```

### *230B. T-primes（选做）



binary search, implementation, math, number theory, 1300, http://codeforces.com/problemset/problem/230/B

思路：

代码

```
import math
 
def sieve_of_eratosthenes(max_num):
    is_prime = [True] * (max_num + 1)
    p = 2
    while (p * p <= max_num):
        if is_prime[p]:
            for i in range(p * p, max_num + 1, p):
                is_prime[i] = False
        p += 1
    primes = [p for p in range(2, max_num + 1) if is_prime[p]]
    return primes
 
n = int(input())
x = list(map(int, input().split()))
 
max_x = max(x)
primes = sieve_of_eratosthenes(int(math.sqrt(max_x)) + 1)
t_primes = {p * p for p in primes}
 
for num in x:
    if num in t_primes:
        print("YES")
    else:
        print("NO")
```

### *12559: 最大最小整数 （选做）



greedy, strings, sortings, http://cs101.openjudge.cn/practice/12559

思路：

代码

```
from functools import cmp_to_key

# 自定义比较函数
def compare_max(x, y):
    # 以拼接后的字符串比较大小
    return (y + x > x + y) - (y + x < x + y)

def compare_min(x, y):
    return (x + y > y + x) - (x + y < y + x)

# 读取输入
n = int(input())
a = input().split()

# 计算最大整数
max_sorted = sorted(a, key=cmp_to_key(compare_max))
max_number = ''.join(max_sorted)

# 计算最小整数
min_sorted = sorted(a, key=cmp_to_key(compare_min))
min_number = ''.join(min_sorted)

# 输出结果
print(max_number, min_number)

```
